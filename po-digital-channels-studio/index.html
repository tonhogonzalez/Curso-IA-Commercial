# Databricks Lakehouse: Cálculo del STP Rate y Time-to-Cash
from pyspark.sql import functions as F

df_funnel = spark.table("commercial_lakehouse.channels.events_lending_funnel")
df_core = spark.table("commercial_lakehouse.core_banking.contracts_loans")

metrics_kpi = (
    df_funnel.groupBy("company_cif")
    .agg(
        F.max(F.when(F.col("step") == "simulated", 1).otherwise(0)).alias("simulated"),
        F.max(F.when(F.col("step") == "multi_signed", 1).otherwise(0)).alias("signed"),
        F.min("ts").alias("t_in"),
        F.max("ts").alias("t_out")
    )
    .join(df_core, on="company_cif", how="left")
    .withColumn("is_pure_stp", F.when((F.col("signed") == 1) & (F.col("branch_assisted") == False), 1).otherwise(0))
    .withColumn("time_to_cash_hours", (F.unix_timestamp("disbursement_ts") - F.unix_timestamp("t_in")) / 3600.0)
)

# Muestra agregada en producción
metrics_kpi.select(
    F.avg("is_pure_stp").alias("stp_ratio"),
    F.percentile_approx("time_to_cash_hours", 0.90).alias("p90_time_to_cash")
).show()
